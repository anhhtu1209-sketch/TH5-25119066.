## 1. docker-compose.yml
```yaml
services:
  mosquitto:
    image: eclipse-mosquitto:2
    container_name: mosquitto
    restart: unless-stopped
    ports:
      - "1883:1883"
    volumes:
      - ./broker/mosquitto.conf:/mosquitto/config/mosquitto.conf
      - mosq-data:/mosquitto/data

  nodered:
    image: nodered/node-red:latest
    container_name: nodered
    restart: unless-stopped
    ports:
      - "1880:1880"
    volumes:
      - nodered-data:/data
    depends_on:
      - mosquitto

volumes:
  mosq-data:
  nodered-data:
```

## 3. device/esp32_mqtt.ino
```cpp
// TH5 - ESP32 -> MQTT broker, tự reconnect + buffer khi mất mạng
// Thư viện: PubSubClient (Nick O'Leary). Board: ESP32 Dev Module.
#include <WiFi.h>
#include <PubSubClient.h>

// ---- Cấu hình: sửa 5 dòng này ----
const char* WIFI_SSID = "YOUR_WIFI";
const char* WIFI_PASS = "YOUR_PASS";
const char* MQTT_HOST = "192.168.1.10";   // IP máy chạy docker compose
const uint16_t MQTT_PORT = 1883;
const char* MSSV = "20xxxxxx";
// -----------------------------------

const uint32_t PUBLISH_MS = 5000;
const uint32_t BACKOFF_MIN = 1000, BACKOFF_MAX = 30000;
const int BUF_N = 50;                      // số mẫu giữ lại khi mất mạng

struct Sample { uint32_t seq; uint32_t upMs; float temp; float hum; };
Sample buf[BUF_N];
int bufHead = 0, bufCount = 0;

WiFiClient net;
PubSubClient mqtt(net);
String topicData, topicStatus, clientId;
uint32_t seqNo = 0, lastPub = 0, lastTry = 0, backoff = BACKOFF_MIN;

Sample readSample() {
  // Thay bằng DHT22/BME280 thật. Đây là dữ liệu mô phỏng.
  Sample s;
  s.seq = seqNo++;
  s.upMs = millis();
  s.temp = 25.0 + (random(-50, 50) / 10.0);
  s.hum = 60.0 + (random(-100, 100) / 10.0);
  return s;
}

void pushSample(const Sample& s) {
  int idx = (bufHead + bufCount) % BUF_N;
  if (bufCount == BUF_N) {                 // đầy: bỏ mẫu cũ nhất
    bufHead = (bufHead + 1) % BUF_N;
    idx = (bufHead + bufCount - 1) % BUF_N;
  } else {
    bufCount++;
  }
  buf[idx] = s;
}

void flushBuffer() {
  while (bufCount > 0 && mqtt.connected()) {
    const Sample& s = buf[bufHead];
    char payload[128];
    snprintf(payload, sizeof(payload),
             "{\"seq\":%lu,\"uptime_ms\":%lu,\"temp\":%.1f,\"hum\":%.1f}",
             (unsigned long)s.seq, (unsigned long)s.upMs, s.temp, s.hum);
    if (!mqtt.publish(topicData.c_str(), payload)) break;  // giữ lại, thử sau
    bufHead = (bufHead + 1) % BUF_N;
    bufCount--;
  }
}

bool tryConnect() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.printf("[%lu] WiFi chua ket noi, thu lai...\n", millis());
    WiFi.disconnect();
    WiFi.begin(WIFI_SSID, WIFI_PASS);
    return false;
  }
  Serial.printf("[%lu] Ket noi MQTT (backoff=%lu ms)...\n", millis(), (unsigned long)backoff);
  // Last Will: broker tự đăng "offline" (retained) khi thiết bị mất kết nối đột ngột
  if (mqtt.connect(clientId.c_str(), NULL, NULL,
                   topicStatus.c_str(), 1, true, "offline")) {
    mqtt.publish(topicStatus.c_str(), "online", true);
    Serial.printf("[%lu] MQTT OK, con %d mau trong buffer\n", millis(), bufCount);
    return true;
  }
  Serial.printf("[%lu] MQTT that bai, state=%d\n", millis(), mqtt.state());
  return false;
}

void setup() {
  Serial.begin(115200);
  clientId = String("esp32-") + MSSV;
  topicData = String("aet2022/") + MSSV + "/sensor";
  topicStatus = String("aet2022/") + MSSV + "/status";

  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);
  WiFi.persistent(false);
  WiFi.begin(WIFI_SSID, WIFI_PASS);

  mqtt.setServer(MQTT_HOST, MQTT_PORT);
  mqtt.setKeepAlive(15);                   // phát hiện mất kết nối trong ~vài chục giây
}

void loop() {
  uint32_t now = millis();

  if (!mqtt.connected()) {
    if (now - lastTry >= backoff) {
      lastTry = now;
      if (tryConnect()) backoff = BACKOFF_MIN;
      else backoff = (backoff * 2 > BACKOFF_MAX) ? BACKOFF_MAX : backoff * 2;
    }
  } else {
    mqtt.loop();
  }

  if (now - lastPub >= PUBLISH_MS) {
    lastPub = now;
    pushSample(readSample());
  }

  if (mqtt.connected()) flushBuffer();
}
```

## 4. README.md
````markdown
# TH5 - ESP32 -> Mosquitto -> Node-RED

## Chạy broker + dashboard
    docker compose up -d
- Node-RED: http://localhost:1880 (dashboard thêm node `node-red-dashboard` qua Manage palette)
- Trong Node-RED: node `mqtt in` -> topic `aet2022/<mssv>/sensor`, broker `mosquitto:1883`
  -> `json` -> `ui_chart` / `ui_gauge`; thêm `mqtt in` cho topic `.../status` -> `ui_text`.

## Firmware
Mở `device/esp32_mqtt.ino`, sửa Wi-Fi, IP máy chạy Docker, MSSV, rồi nạp.

## Thử mất mạng
1. Tắt Wi-Fi 30-60 s rồi bật lại.
2. `docker stop mosquitto` rồi `docker start mosquitto`.
Lưu log Serial (có timestamp) vào `tests/` và chụp dashboard.
Trường `seq` trong payload cho thấy mẫu nào được gửi bù sau khi nối lại.
````
